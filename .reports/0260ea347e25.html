WITH recent_orders AS (
  SELECT o.order_id, o.order_date, COALESCE(o.freight, 0)::numeric AS freight
  FROM public.orders o
  WHERE o.order_date >= (
    SELECT MAX(order_date) - INTERVAL '29 days' FROM public.orders
  )
    AND o.order_date <= (
    SELECT MAX(order_date) FROM public.orders
  )
), order_totals AS (
  SELECT ro.order_id,
         ro.order_date,
         SUM((od.unit_price::numeric * od.quantity::numeric) * (1 - od.discount::numeric)) AS line_revenue,
         ro.freight
  FROM recent_orders ro
  JOIN public.order_details od ON od.order_id = ro.order_id
  GROUP BY ro.order_id, ro.order_date, ro.freight
)
SELECT order_date,
       COUNT(*) AS orders_count,
       ROUND(SUM(line_revenue + freight), 2) AS daily_revenue,
       ROUND(AVG(line_revenue + freight), 2) AS avg_order_value
FROM order_totals
GROUP BY order_date
ORDER BY order_date

WITH recent_orders AS (
  SELECT o.order_id, o.order_date, COALESCE(o.freight, 0)::numeric AS freight
  FROM public.orders o
  WHERE o.order_date >= (
    SELECT MAX(order_date) - INTERVAL '29 days' FROM public.orders
  )
    AND o.order_date <= (
    SELECT MAX(order_date) FROM public.orders
  )
), order_totals AS (
  SELECT ro.order_id,
         ro.order_date,
         SUM((od.unit_price::numeric * od.quantity::numeric) * (1 - od.discount::numeric)) AS line_revenue,
         ro.freight,
         SUM((od.unit_price::numeric * od.quantity::numeric) * (1 - od.discount::numeric)) + ro.freight AS order_revenue
  FROM recent_orders ro
  JOIN public.order_details od ON od.order_id = ro.order_id
  GROUP BY ro.order_id, ro.order_date, ro.freight
)
SELECT order_id,
       order_date,
       ROUND(order_revenue, 2) AS order_revenue
FROM order_totals
ORDER BY order_revenue DESC, order_id
LIMIT 5
